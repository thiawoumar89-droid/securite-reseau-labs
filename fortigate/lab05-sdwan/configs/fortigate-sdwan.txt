# FortiGate VM 7.4.12 — Lab 05 SD-WAN (extraits nettoyés, sans secrets)

config system interface
    edit port1
        set mode static
        set ip 192.168.122.21 255.255.255.0
        set alias WAN1
        set role wan
        set allowaccess ping
    next
    edit port3
        set mode static
        set ip 100.64.2.1 255.255.255.252
        set alias WAN2
        set role wan
        set allowaccess ping
    next
end

config system dns
    set primary 8.8.8.8
    set secondary 192.168.122.1
end

config firewall address
    edit "Cloudflare-DNS"
        set subnet 1.1.1.1 255.255.255.255
    next
end

config system sdwan
    set status enable
    config zone
        edit "virtual-wan-link"
        next
    end
    config members
        edit 1
            set interface "port1"
            set zone "virtual-wan-link"
            set gateway 192.168.122.1
        next
        edit 2
            set interface "port3"
            set zone "virtual-wan-link"
            set gateway 100.64.2.2
        next
    end
    config health-check
        edit "SLA-Internet"
            set server "8.8.8.8"
            set protocol ping
            set interval 500
            set failtime 3
            set recoverytime 3
            set members 1 2
            config sla
                edit 1
                    set latency-threshold 150
                    set jitter-threshold 50
                    set packetloss-threshold 5
                next
            end
        next
    end
    config service
        edit 1
            set name "Cloudflare-par-WAN2"
            set dst "Cloudflare-DNS"
            set src "all"
            set priority-members 2
        next
        edit 2
            set name "Qualite-prefere-WAN1"
            set mode sla
            set dst "all"
            set src "all"
            config sla
                edit "SLA-Internet"
                    set id 1
                next
            end
            set priority-members 1 2
        next
    end
end

config router static
    edit 1
        set sdwan-zone "virtual-wan-link"
    next
end

config firewall policy
    edit 1
        set name "LAN-to-SDWAN"
        set srcintf "port2"
        set dstintf "virtual-wan-link"
        set srcaddr "all"
        set dstaddr "all"
        set action accept
        set schedule "always"
        set service "ALL"
        set nat enable
        set logtraffic all
    next
end
